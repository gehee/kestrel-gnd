# The software frontend carried forward from the Ascent bring-up branch.
# It draws normalized state; native radio/media ownership stays outside it.
include(FetchContent)
include("${PROJECT_SOURCE_DIR}/cmake/ui_idle_assets.cmake")
set(KESTREL_LVGL_SOURCE_DIR "" CACHE PATH "Existing LVGL 9.2.2 source tree")
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
if(KESTREL_LVGL_SOURCE_DIR)
    add_subdirectory("${KESTREL_LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
else()
    FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git
        GIT_TAG 7f07a129e8d77f4984fff8e623fd5be18ff42e74)
    FetchContent_MakeAvailable(lvgl)
endif()
add_library(kg_lvgl_core STATIC model.cpp controller.cpp hud_data.cpp font_atlas.cpp
    input.cpp preferences.cpp ../hud_theme.cpp hud_protocol.cpp hud_buffer.cpp)
target_include_directories(kg_lvgl_core PUBLIC "${PROJECT_SOURCE_DIR}")
target_compile_options(kg_lvgl_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(kg_lvgl_view STATIC hud_view.cpp vector_hud.cpp hud_renderer.cpp
    perspective_text.cpp hud_source.cpp display.cpp view.cpp idle_scene.cpp)
target_link_libraries(kg_lvgl_view PUBLIC kg_lvgl_core lvgl m pthread)
if(TARGET kestrel-idle-assets)
    add_dependencies(kg_lvgl_view kestrel-idle-assets)
endif()
target_compile_options(kg_lvgl_view PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kestrel-hud hud_main.cpp)
target_link_libraries(kestrel-hud PRIVATE kg_lvgl_view)
add_executable(kestrel-hud-vector hud_main.cpp)
target_compile_definitions(kestrel-hud-vector PRIVATE KESTREL_VECTOR_HUD=1)
target_link_libraries(kestrel-hud-vector PRIVATE kg_lvgl_view)
add_executable(kestrel-ui-assetgen assetgen.cpp)
target_include_directories(kestrel-ui-assetgen PRIVATE "${PROJECT_SOURCE_DIR}")
add_executable(kestrel-ui-animation-bench animation_bench.cpp)
target_link_libraries(kestrel-ui-animation-bench PRIVATE kg_lvgl_view)
add_executable(kestrel-ui-shot ui_shot.cpp)
target_link_libraries(kestrel-ui-shot PRIVATE kg_lvgl_view)
if(KESTREL_TESTS)
    add_executable(lvgl_core_test tests.cpp)
    target_link_libraries(lvgl_core_test PRIVATE kg_lvgl_core)
    add_test(NAME lvgl_core COMMAND lvgl_core_test)
    add_executable(lvgl_input_test input_tests.cpp)
    target_link_libraries(lvgl_input_test PRIVATE kg_lvgl_core)
    add_test(NAME lvgl_input COMMAND lvgl_input_test)
    add_executable(lvgl_render_test render_tests.cpp)
    target_link_libraries(lvgl_render_test PRIVATE kg_lvgl_view)
    add_executable(lvgl_hud_test hud_tests.cpp)
    target_link_libraries(lvgl_hud_test PRIVATE kg_lvgl_view)
    set(KESTREL_UI_TEST_ASSETS "" CACHE PATH "Generated software HUD test assets")
    if(KESTREL_UI_TEST_ASSETS)
        add_test(NAME lvgl_render COMMAND lvgl_render_test "${KESTREL_UI_TEST_ASSETS}")
        add_test(NAME lvgl_hud COMMAND lvgl_hud_test "${KESTREL_UI_TEST_ASSETS}" "$<TARGET_FILE:kestrel-hud>")
        add_test(NAME lvgl_vector COMMAND lvgl_hud_test "${KESTREL_UI_TEST_ASSETS}" "$<TARGET_FILE:kestrel-hud-vector>" vector)
    endif()
endif()
